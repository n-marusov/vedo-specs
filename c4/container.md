    System_Boundary(vedo, "VEDO Core") {
        Container(spa, "Vue 3 SPA", "TypeScript/Vue 3", "Веб-интерфейс")

        Container(ontology, "Ontology Service", "Rust/Actix-web", "Graph operations, TBox/ABox, Neo4j driver")
        Container(documentExtractor, "Document Extractor", "Python", "Извлечение OWL-онтологии из документов через LLM: парсинг форматов, генерация последовательности шагов, предпросмотр")
        Container(versioning, "Versioning Service", "Rust/Tokio", "Git-like versioning")
        Container(auth, "Auth Service", "Go/gin", "JWT, OAuth2")
        Container(metrics, "Metrics Service", "Python/FastAPI", "Metrics collection")
        Container(publisher, "Publisher Service", "Rust/Actix-web", "Snapshot creation, publication CRUD")
        Container(publishBrowse, "Browse UI", "Vue 3 SPA", "Публичная навигация и поиск")
        Container(publicBrowseApi, "Public Browse API", "Rust/Actix-web", "Неавторизованные read-only endpoints для навигации и поиска")
        Container(cli, "vedo-cli", "Go", "CLI-утилита администрирования: backup/restore, diagnose, emergency, decommission, airgap, MR, support (Single binary, не контейнеризируется)")
        Container(commenting, "Commenting Service", "Go/gorilla-websocket", "Комментарии к сущностям онтологии с real-time доставкой через WebSocket")
        Container(aiOrch, "AI Orchestration Service", "Go/gRPC", "AI-оркестрация: NL→OWL, NL→Query, подсказки, рефайнмент, LLM Policy Router, Prompt Injection защита, CheckPolicy/LogLLMUsage")
        Container(ticketApi, "Ticket API", "Go/REST", "Создание, обновление, просмотр тикетов")
        Container(ticketTelemetry, "Telemetry Listener", "Go/Kafka или HTTP", "Автоматическое создание тикетов по телеметрии")

        Container(ticketSync, "Ticket Synchronizer", "Go", "Двусторонняя синхронизация с внешней системой тикетов")
        Container(ticketNotifier, "Ticket Notifier", "Go", "Уведомления: email, in-app, эскалации P0/P1")
        ContainerDb(neo4j, "Neo4j Cluster", "Cypher", "3 core + 2 read replica, рабочая онтология")
        ContainerDb(publishNeo4j, "Public Neo4j", "Cypher", "Read-only копия для публичного доступа")
        ContainerDb(postgres, "PostgreSQL Cluster", "SQL/JSONB", "Primary + 2 standby (Version Store)")
        ContainerDb(supportDb, "Support DB", "PostgreSQL", "Отдельный кластер: support metadata, tenant info, break-glass audit")
        ContainerDb(commentingDb, "Commenting DB", "PostgreSQL", "Комментарии, нити, метаданные")
        ContainerDb(ticketDb, "Ticket DB", "PostgreSQL", "Тикеты, история, метаданные, синхронизация")
        ContainerDb(redis, "Redis Cluster", "RESP", "3 master + 3 replica")
        ContainerDb(rabbitmq, "RabbitMQ", "AMQP", "Events")
        ContainerDb(replicaStore, "Хранилище реплик", "S3/MinIO-compatible", "WAL, snapshots, off-site copy")
        ContainerDb(immutableBackup, "Immutable Backup Storage", "S3/MinIO + WORM/Object Lock", "Immutable backup копия, изолированные credentials")
        ContainerDb(vault, "Vault / KMS", "HashiCorp Vault / encFS", "Key management, secrets, encryption keys")
    }

    System_Ext(keycloak, "Keycloak", "Identity Provider")
    System_Ext(industry, "Отраслевая система", "ERP/PLM/MES")
